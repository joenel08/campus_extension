ALTER TABLE `templates`
    MODIFY COLUMN `template_code` VARCHAR(100) 
    NULL DEFAULT NULL 
    COLLATE 'utf8mb4_0900_ai_ci';


CREATE TABLE `detailed_proposals` (
  `id` INT(10) NOT NULL AUTO_INCREMENT,
  `submission_id` INT(10) NOT NULL,          -- FK to submissions (the parent proposal)
  `user_id` INT(10) NOT NULL,
  `academic_year_id` INT(10) NULL DEFAULT NULL,
  `attachment` VARCHAR(255) NULL DEFAULT NULL,
  `status` ENUM('draft','submitted','approved','revision','rejected') NOT NULL DEFAULT 'draft',
  `admin_remarks` TEXT NULL DEFAULT NULL,
  `viewed_at` TIMESTAMP NULL DEFAULT NULL,
  `created_at` TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at` TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uniq_submission` (`submission_id`),
  INDEX `user_id` (`user_id`),
  INDEX `fk_dp_ay` (`academic_year_id`),
  CONSTRAINT `fk_dp_submission` FOREIGN KEY (`submission_id`) REFERENCES `submissions` (`id`) ON DELETE CASCADE,
  CONSTRAINT `fk_dp_user` FOREIGN KEY (`user_id`) REFERENCES `users` (`id`) ON DELETE CASCADE,
  CONSTRAINT `fk_dp_ay` FOREIGN KEY (`academic_year_id`) REFERENCES `academic_years` (`id`) ON DELETE SET NULL
) ENGINE=InnoDB;
